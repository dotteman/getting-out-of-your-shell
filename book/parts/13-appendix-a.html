Scripting skeleton

#!/usr/bin/env bash
set -euo pipefail

target="${1:?usage: $0 TARGET}"
tmpdir=$(mktemp -d)
trap 'rm -rf "$tmpdir"' EXIT

log() { echo "[$(date +'%F %T')] $*" >&2; }

if [[ -d "$target" ]]; then
  for f in "$target"/*; do
    [[ -f "$f" ]] || continue
    log "processing ${f##*/}"
  done
else
  log "not a directory: $target"
  exit 1
fi
